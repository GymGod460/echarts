### 扩展组件
```
series:[{
    type:'bar',
    itemStyle:{ // 柱样式
        color:'#3398db',
        barBorderRadius:[4,4,0,0] // 柱圆角
    },
    label:{show:true}, // 常态直接显示数值（非悬浮）
    barCategoryGap:'20%', // 不同类目之间空隙
    markPoint:{ // 标记点，最大值最小值标记
        data:[{type:'max'},{type:'min'}]
    },
    silent:false, // 是否关闭鼠标交互
}]
```
